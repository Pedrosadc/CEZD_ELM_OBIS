interval:
  - interval: 200ms
    then:
      - lambda: |-
          using namespace esphome;

          static uint8_t frame[1024];
          static int pos = 0;
          static unsigned long last = millis();
          uint8_t b;

          while (id(han_uart).read_byte(&b)) {
            if (pos < 1023) frame[pos++] = b;
            last = millis();
          }

          if (pos > 0 && millis() - last > 200) {

            ESP_LOGI("han", "Complete frame (%d bytes)", pos);

            struct ObisDef {
              uint8_t obis[6];
              uint8_t index;
              float scale;
            };

            static const ObisDef obis_list[] = {
              {{0x01,0x00,0x01,0x07,0x00,0xFF},  0, 1.0f},
              {{0x01,0x00,0x02,0x07,0x00,0xFF},  1, 1.0f},
              {{0x01,0x00,0x01,0x08,0x00,0xFF},  2, 1.0f},
              {{0x01,0x00,0x02,0x08,0x00,0xFF},  3, 1.0f},
              {{0x01,0x00,0x15,0x07,0x00,0xFF}, 10, 1.0f},
              {{0x01,0x00,0x29,0x07,0x00,0xFF}, 11, 1.0f},
              {{0x01,0x00,0x3D,0x07,0x00,0xFF}, 12, 1.0f},
              {{0x01,0x00,0x01,0x08,0x01,0xFF}, 13, 1.0f},
              {{0x01,0x00,0x01,0x08,0x02,0xFF}, 14, 1.0f},
              {{0x01,0x00,0x01,0x08,0x03,0xFF}, 15, 1.0f},
              {{0x01,0x00,0x01,0x08,0x04,0xFF}, 16, 1.0f},
              {{0x01,0x00,0x02,0x08,0x01,0xFF}, 17, 1.0f},
              {{0x01,0x00,0x02,0x08,0x02,0xFF}, 18, 1.0f},
              {{0x01,0x00,0x02,0x08,0x03,0xFF}, 19, 1.0f},
              {{0x01,0x00,0x02,0x08,0x04,0xFF}, 20, 1.0f},
              {{0x00,0x00,0x11,0x00,0x00,0xFF}, 21, 1.0f},
              {{0x00,0x60,0x0E,0x00,0xFF,0x02}, 22, 1.0f},
              {{0x00,0x00,0x60,0x01,0x01,0xFF}, 23, 1.0f},
              {{0x00,0x00,0x60,0x03,0x0A,0xFF}, 24, 1.0f},
              {{0x00,0x01,0x60,0x03,0x0A,0xFF}, 25, 1.0f},
              {{0x00,0x02,0x60,0x03,0x0A,0xFF}, 26, 1.0f},
              {{0x00,0x03,0x60,0x03,0x0A,0xFF}, 27, 1.0f},
              {{0x00,0x04,0x60,0x03,0x0A,0xFF}, 28, 1.0f},
            };

            auto match_obis = [&](int start, const uint8_t obis[6]) -> bool {
              for (int k = 0; k < 6; k++)
                if (frame[start + k] != obis[k]) return false;
              return true;
            };

            auto read_u32 = [&](int p) -> uint32_t {
              return (uint32_t(frame[p]) << 24) |
                     (uint32_t(frame[p+1]) << 16) |
                     (uint32_t(frame[p+2]) << 8) |
                     frame[p+3];
            };

            for (int i = 0; i < pos - 12; i++) {
              for (auto &def : obis_list) {
                if (!match_obis(i, def.obis)) continue;

                if (def.index == 22) {
                  char c1 = (char)frame[i+8];
                  char c2 = (char)frame[i+9];
                  if (c1 == 'T' && c2 >= '1' && c2 <= '4') {
                    id(tarif_state_pico).publish_state((float)(c2 - '0'));
                  }
                  continue;
                }

                if (def.index == 23) {
                  int len = frame[i+8];
                  std::string meter;
                  for (int k = 0; k < len; k++) meter.push_back((char)frame[i+9+k]);
                  id(meter_number_pico).publish_state(meter);
                  continue;
                }

                float value = read_u32(i + 8) * def.scale;

                switch (def.index) {
                  case 0:  id(p_plus_pico).publish_state(value); break;
                  case 1:  id(p_minus_pico).publish_state(value); break;
                  case 2:  id(a_plus_pico).publish_state(value); break;
                  case 3:  id(a_minus_pico).publish_state(value); break;
                  case 10: id(p_l1_pico).publish_state(value); break;
                  case 11: id(p_l2_pico).publish_state(value); break;
                  case 12: id(p_l3_pico).publish_state(value); break;
                  case 13: id(a_plus_t1_pico).publish_state(value); break;
                  case 14: id(a_plus_t2_pico).publish_state(value); break;
                  case 15: id(a_plus_t3_pico).publish_state(value); break;
                  case 16: id(a_plus_t4_pico).publish_state(value); break;
                  case 17: id(a_minus_t1_pico).publish_state(value); break;
                  case 18: id(a_minus_t2_pico).publish_state(value); break;
                  case 19: id(a_minus_t3_pico).publish_state(value); break;
                  case 20: id(a_minus_t4_pico).publish_state(value); break;
                  case 21: id(limiter_pico).publish_state(value); break;
                  case 24: id(disconnector_state_pico).publish_state(frame[i+8]); break;
                  case 25: id(relay1_state_pico).publish_state(frame[i+8]); break;
                  case 26: id(relay2_state_pico).publish_state(frame[i+8]); break;
                  case 27: id(relay3_state_pico).publish_state(frame[i+8]); break;
                  case 28: id(relay4_state_pico).publish_state(frame[i+8]); break;
                }
              }
            }

            // === PŘÍMÝ VÝPOČET A PUBLIKACE ODVOZENÝCH HODNOT ===
            if (!isnan(id(p_plus_pico).state) && !isnan(id(p_minus_pico).state)) {
              id(p_total_pico).publish_state(id(p_plus_pico).state - id(p_minus_pico).state);
            }
            if (!isnan(id(p_l1_pico).state) && !isnan(id(p_l2_pico).state) && !isnan(id(p_l3_pico).state)) {
              id(p_sum_pico).publish_state(id(p_l1_pico).state + id(p_l2_pico).state + id(p_l3_pico).state);
            }
            if (!isnan(id(a_plus_pico).state)) {
              id(a_plus_kwh_pico).publish_state(id(a_plus_pico).state / 1000.0f);
            }
            if (!isnan(id(a_minus_pico).state)) {
              id(a_minus_kwh_pico).publish_state(id(a_minus_pico).state / 1000.0f);
            }
            
            // Přepočty kWh pro T1-T4 Odběr (A+)
            if (!isnan(id(a_plus_t1_pico).state)) id(a_plus_t1_kwh_pico).publish_state(id(a_plus_t1_pico).state / 1000.0f);
            if (!isnan(id(a_plus_t2_pico).state)) id(a_plus_t2_kwh_pico).publish_state(id(a_plus_t2_pico).state / 1000.0f);
            if (!isnan(id(a_plus_t3_pico).state)) id(a_plus_t3_kwh_pico).publish_state(id(a_plus_t3_pico).state / 1000.0f);
            if (!isnan(id(a_plus_t4_pico).state)) id(a_plus_t4_kwh_pico).publish_state(id(a_plus_t4_pico).state / 1000.0f);

            // Přepočty kWh pro T1-T4 Dodávka (A-)
            if (!isnan(id(a_minus_t1_pico).state)) id(a_minus_t1_kwh_pico).publish_state(id(a_minus_t1_pico).state / 1000.0f);
            if (!isnan(id(a_minus_t2_pico).state)) id(a_minus_t2_kwh_pico).publish_state(id(a_minus_t2_pico).state / 1000.0f);
            if (!isnan(id(a_minus_t3_pico).state)) id(a_minus_t3_kwh_pico).publish_state(id(a_minus_t3_pico).state / 1000.0f);
            if (!isnan(id(a_minus_t4_pico).state)) id(a_minus_t4_kwh_pico).publish_state(id(a_minus_t4_pico).state / 1000.0f);

            // Binární stavy odpojovače a relé
            if (!isnan(id(disconnector_state_pico).state)) {
              id(disconnector_binary_pico).publish_state(id(disconnector_state_pico).state == 1);
            }
            if (!isnan(id(relay1_state_pico).state)) id(relay1_binary_pico).publish_state(id(relay1_state_pico).state == 1);
            if (!isnan(id(relay2_state_pico).state)) id(relay2_binary_pico).publish_state(id(relay2_state_pico).state == 1);
            if (!isnan(id(relay3_state_pico).state)) id(relay3_binary_pico).publish_state(id(relay3_state_pico).state == 1);
            if (!isnan(id(relay4_state_pico).state)) id(relay4_binary_pico).publish_state(id(relay4_state_pico).state == 1);

            // Textový popis tarifu
            if (!isnan(id(tarif_state_pico).state)) {
              int t = (int) id(tarif_state_pico).state;
              if (t == 1) id(tarif_text_pico).publish_state("T1 – VT");
              else if (t == 2) id(tarif_text_pico).publish_state("T2 – NT");
              else if (t == 3) id(tarif_text_pico).publish_state("T3 – Speciální tarif");
              else if (t == 4) id(tarif_text_pico).publish_state("T4 – Speciální tarif");
              else id(tarif_text_pico).publish_state("Neznámý");
            }

            pos = 0;
          }
